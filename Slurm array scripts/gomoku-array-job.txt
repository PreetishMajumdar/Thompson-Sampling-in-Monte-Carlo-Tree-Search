#!/bin/bash --login
#SBATCH -p serial
#SBATCH -n 1
#SBATCH -t 0-06:00:00
#SBATCH --job-name=mcts-gomoku-array    # 1. Changed Job Name
#SBATCH --output=logs/mcts-gomoku-%A-%a.out
#SBATCH --array=0-9

START_ID=$(( SLURM_ARRAY_TASK_ID * 10 + 1 ))

echo ""
echo "Slurm Array Job ID: $SLURM_ARRAY_JOB_ID"
echo "Array Task Index:   $SLURM_ARRAY_TASK_ID"
echo "Running Game IDs:   $START_ID to $(( START_ID + 9 ))"
echo "Started at: $(date)"
echo ""

module purge
module load apps/binapps/conda/miniforge3/25.9.1
source activate mcts-env

# Run this chunk of games
python tournament.py \
    --game gomoku \
    --board_size 15 \
    --iterations 5000 \
    --start_id $START_ID \
    --chunk_size 10

echo "Finished at: $(date)"
